cmake_minimum_required(VERSION 3.16)
project(e4eg_probe LANGUAGES C CXX)

# The source is supplied explicitly; configuration never downloads a dependency.
set(LLAMA_SOURCE "" CACHE PATH "Clean llama.cpp checkout at the pinned commit")
set(LLAMA_PIN "1f8fa52318b594b373e7447a3c2bb36cdba9515f")
if(NOT LLAMA_SOURCE OR NOT EXISTS "${LLAMA_SOURCE}/include/llama.h")
  message(FATAL_ERROR "Set LLAMA_SOURCE to the pinned llama.cpp checkout")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse HEAD WORKING_DIRECTORY "${LLAMA_SOURCE}"
                RESULT_VARIABLE git_result OUTPUT_VARIABLE revision OUTPUT_STRIP_TRAILING_WHITESPACE)
execute_process(COMMAND "${GIT_EXECUTABLE}" status --porcelain --untracked-files=normal
                WORKING_DIRECTORY "${LLAMA_SOURCE}" RESULT_VARIABLE status_result
                OUTPUT_VARIABLE dirty OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT git_result EQUAL 0 OR NOT status_result EQUAL 0 OR NOT revision STREQUAL LLAMA_PIN OR dirty)
  message(FATAL_ERROR "llama.cpp must be a clean checkout at ${LLAMA_PIN}")
endif()
if(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
  message(FATAL_ERROR "Use a single-configuration Release build")
endif()
# Force a static CPU-only build. No downloaded libraries, GPU, RPC, BLAS or OpenMP.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(option GGML_CUDA GGML_VULKAN GGML_METAL GGML_SYCL GGML_RPC GGML_BLAS GGML_OPENMP GGML_NATIVE GGML_BACKEND_DL GGML_OPENCL GGML_CANN GGML_HIP GGML_MUSA GGML_KOMPUTE GGML_HEXAGON GGML_WEBGPU GGML_ZDNN GGML_VIRTGPU GGML_VIRTGPU_BACKEND GGML_OPENVINO GGML_ET GGML_ZENDNN GGML_ACCELERATE GGML_CPU_ALL_VARIANTS)
  set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(option LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_SERVER LLAMA_BUILD_APP)
  set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${LLAMA_SOURCE}" "${CMAKE_BINARY_DIR}/llama" EXCLUDE_FROM_ALL)
# Embed the configured compiler and CPU options as provenance, not as a determinism claim.
set(E4EG_CPU_INFO "cmake=${CMAKE_VERSION},system=${CMAKE_SYSTEM_NAME},processor=${CMAKE_SYSTEM_PROCESSOR}")
foreach(opt GGML_NATIVE GGML_LLAMAFILE GGML_CPU_REPACK GGML_SSE42 GGML_F16C GGML_FMA GGML_BMI2 GGML_AVX GGML_AVX2 GGML_AVX512 GGML_AVX512_VBMI GGML_AVX512_VNNI GGML_AMX_TILE GGML_AMX_INT8 GGML_AMX_BF16)
  string(APPEND E4EG_CPU_INFO ",${opt}=${${opt}}")
endforeach()
foreach(var CMAKE_C_COMPILER_ID CMAKE_C_COMPILER_VERSION CMAKE_CXX_COMPILER_ID CMAKE_CXX_COMPILER_VERSION CMAKE_C_FLAGS CMAKE_C_FLAGS_RELEASE CMAKE_CXX_FLAGS CMAKE_CXX_FLAGS_RELEASE E4EG_CPU_INFO)
  string(REPLACE "\\" "\\\\" ${var}_ESC "${${var}}")
  string(REPLACE "\"" "\\\"" ${var}_ESC "${${var}_ESC}")
endforeach()
configure_file(build_info.h.in build_info.h @ONLY)
add_executable(e4eg-native native.cpp)
target_compile_features(e4eg-native PRIVATE cxx_std_17)
target_include_directories(e4eg-native PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(e4eg-native PRIVATE llama)
enable_testing()
add_executable(e4eg-core-test tests/core_test.cpp)
target_compile_features(e4eg-core-test PRIVATE cxx_std_17)
add_test(NAME e4eg-core-test COMMAND e4eg-core-test)
add_test(NAME e4eg-native-default COMMAND e4eg-native)
